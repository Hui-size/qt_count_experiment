cmake_minimum_required(VERSION 3.21)
project(QtCountExperiment VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_SOURCE_DIR}")

find_package(Qt6 6.10.3 REQUIRED COMPONENTS Widgets)
add_library(calculator_engine STATIC calculatorengine.cpp calculatorengine.h)
target_include_directories(calculator_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(calculator_engine PUBLIC Qt6::Core)
add_library(calculator_widgets STATIC
    mainwindow.cpp mainwindow.h mainwindow.ui)
target_include_directories(calculator_widgets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(calculator_widgets PUBLIC Qt6::Widgets calculator_engine)
add_executable(qt_calculator WIN32 main.cpp)
target_link_libraries(qt_calculator PRIVATE calculator_widgets)
if(MINGW)
    target_compile_options(calculator_widgets PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(calculator_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(test_calculator tests/test_calculator.cpp)
    target_link_libraries(test_calculator PRIVATE calculator_widgets Qt6::Test)
    add_test(NAME calculator_ui COMMAND test_calculator)
    set_tests_properties(calculator_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(test_engine tests/test_engine.cpp)
    target_link_libraries(test_engine PRIVATE calculator_engine Qt6::Test)
    add_test(NAME calculator_engine COMMAND test_engine)
endif()
