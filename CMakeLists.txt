cmake_minimum_required(VERSION 3.21)
project(QtCountExperiment VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_SOURCE_DIR}/src")

find_package(Qt6 6.9 REQUIRED COMPONENTS Widgets)
add_library(calculator_widgets STATIC
    src/calculatorwindow.cpp src/calculatorwindow.h src/calculatorwindow.ui)
target_include_directories(calculator_widgets PUBLIC src)
target_link_libraries(calculator_widgets PUBLIC Qt6::Widgets)
add_executable(qt_calculator WIN32 src/main.cpp)
target_link_libraries(qt_calculator PRIVATE calculator_widgets)
if(MINGW)
    target_compile_options(calculator_widgets PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(test_calculator tests/test_calculator.cpp)
    target_link_libraries(test_calculator PRIVATE calculator_widgets Qt6::Test)
    add_test(NAME calculator_ui COMMAND test_calculator)
    set_tests_properties(calculator_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
